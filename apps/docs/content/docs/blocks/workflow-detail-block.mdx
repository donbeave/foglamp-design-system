## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `WorkflowDetailBlock` | `nameParam?` | `string` | Active workflow name parameter. |
|  | `data?` | `WorkflowDetailData` | Structured dataset rendered by the block. |
|  | `isLoading?` | `boolean` | Renders loading skeleton placeholders while data is being fetched. |
|  | `isUpdating?` | `boolean` | Holds the current view while refreshed data is loading. |
|  | `initialRun?` | `string \| null` | Initially selected workflow run identifier. |
|  | `onNavigate?` | `(href: string) => void` | Callback fired when navigating to a linked trace, agent, or workflow. |
|  | `onRunChange?` | `(id: string \| null) => void` | Callback fired when the selected workflow run changes. |

## Usage

```tsx
import { useState } from "react";
import {
  WorkflowDetailBlock,
} from "@tailrocks/framerock/blocks/dashboard/workflow-detail-block";

export function WorkflowDetailExample({
  nameParam = 'Support',
  isUpdating = false,
}: {
  nameParam?: string;
  isUpdating?: boolean;
} = {}) {
  const [run, setRun] = useState<string | null>(null);
  return (
    <WorkflowDetailBlock
      nameParam={nameParam}
      isUpdating={isUpdating}
      initialRun={run}
      onRunChange={setRun}
      onNavigate={() => {}}
    />
  );
}
```
