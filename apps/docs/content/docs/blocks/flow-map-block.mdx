## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `FlowMap` | `graph` | `ScanData["graph"]` | Scan architecture graph containing nodes, edges, and groups. |
|  | `focusKinds` | `NodeKind[] \| null` | Optional node kind filter spotlighting matching nodes on the canvas. |
|  | `embedded?` | `boolean` | Renders compact marketing embed mode with tight fit and no pan/zoom controls. |
|  | `frame?` | `boolean` | Renders interactive iframe embed mode fitted without sidebar clearance. |
|  | `direction?` | `"RIGHT" \| "DOWN"` | ELK layout flow direction (`"RIGHT"` for landscape or `"DOWN"` for portrait). |
|  | `padding?` | `{ left?: number; right?: number }` | Viewport fit clearance in pixels around the laid-out graph. |
|  | `workflowGroups?` | `string[]` | Group labels styled with the emerald workflow badge treatment. |
|  | `focusGroups?` | `string[] \| null` | Group labels spotlighted on the canvas while dimming other nodes. |
|  | `focusLabels?` | `string[] \| null` | Node labels spotlighted on the canvas while dimming other nodes. |
|  | `zoomTo?` | `MapZoomTarget \| null` | Camera target framing the specified node or group labels. |

## Usage

```tsx
import { FlowMap } from "@tailrocks/framerock/blocks/scan/flow-map";
import {
  ScanFaviconProvider,
} from "@tailrocks/framerock/blocks/scan/favicon-context";
import { NodeKind } from "@tailrocks/framerock/lib/scan/scan";
import type { ScanData } from "@tailrocks/framerock/lib/scan/scan";
import { getGoogleFavicon } from "@tailrocks/framerock/widgets/leaf/favicon";

const scanFixture: ScanData = {
  version: 1,
  project: {
    name: "Framerock",
    slug: "framerock",
    tagline: "AI observability, clearly mapped",
    iconDomain: "tailrocks.com",
    date: "2026-11-05",
  },
  stats: { agents: 2, models: 2, tools: 2, integrations: 1 },
  topModels: [{ id: "gpt-4o", label: "GPT-4o", domain: "openai.com" }],
  topTools: [{ id: "search", label: "Search", domain: "exa.ai" }],
  topIntegrations: [{ id: "slack", label: "Slack", domain: "slack.com" }],
  graph: {
    nodes: [
      { id: "webhook", label: "Incoming webhook", kind: "entry" },
      { id: "support", label: "Support agent", kind: "agent" },
      { id: "openai", label: "GPT-4o", kind: "model", domain: "openai.com" },
    ],
    edges: [
      { from: "webhook", to: "support", kind: "triggers" },
      { from: "support", to: "openai", kind: "calls" },
    ],
  },
};

const resolveStoryFavicon = getGoogleFavicon;

export function FlowMapExample({
  direction = 'RIGHT',
}: {
  direction?: 'RIGHT' | 'DOWN';
} = {}) {
  return (
    <ScanFaviconProvider resolve={resolveStoryFavicon}>
      <div
        className="relative h-[640px] w-full overflow-hidden rounded-xl border border-border/60 bg-background"
      >
        <FlowMap
          graph={scanFixture.graph}
          focusKinds={null as NodeKind[] | null}
          direction={direction}
          padding={{ left: 48, right: 48 }}
        />
      </div>
    </ScanFaviconProvider>
  );
}
```
