## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `AgentDetailInspectorBlock` | `agentName?` | `string` | Active agent name displayed in the inspector header. |
|  | `data?` | `AgentDetailData` | Structured dataset rendered by the block. |
|  | `isLoading?` | `boolean` | Renders loading skeleton placeholders while data is being fetched. |
|  | `isUpdating?` | `boolean` | Holds the current view while refreshed data is loading. |
|  | `initialTrace?` | `string \| null` | Initially selected trace identifier in the drawer. |
|  | `onNavigate?` | `(href: string) => void` | Callback fired when navigating to a linked trace, agent, or workflow. |
|  | `onTraceChange?` | `(id: string \| null) => void` | Callback fired when the inspected trace changes. |

## Usage

```tsx
import { useState } from "react";
import {
  AgentDetailInspectorBlock,
} from "@tailrocks/framerock/blocks/dashboard/agent-detail-inspector-block";

export function AgentDetailExample({
  agentName = 'Support agent',
  isUpdating = false,
}: {
  agentName?: string;
  isUpdating?: boolean;
} = {}) {
  const [trace, setTrace] = useState<string | null>(null);
  return (
    <AgentDetailInspectorBlock
      agentName={agentName}
      isUpdating={isUpdating}
      initialTrace={trace}
      onTraceChange={setTrace}
      onNavigate={() => {}}
    />
  );
}
```
