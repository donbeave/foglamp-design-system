## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `SessionConversationInspectorBlock` | `sessionId?` | `string` | Active session identifier. |
|  | `data?` | `SessionDetailData \| null` | Structured dataset rendered by the block. |
|  | `isLoading?` | `boolean` | Renders loading skeleton placeholders while data is being fetched. |

## Usage

```tsx
import {
  SessionConversationInspectorBlock,
} from "@tailrocks/framerock/blocks/dashboard/session-conversation-inspector-block";

export function SessionConversationExample({
  sessionId = 'ses_a91f',
  isLoading = false,
}: {
  sessionId?: string;
  isLoading?: boolean;
} = {}) {
  return (
    <SessionConversationInspectorBlock
      sessionId={sessionId}
      isLoading={isLoading}
    />
  );
}
```
