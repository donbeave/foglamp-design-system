## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `ScanLanding` | `prompt?` | `string` | Onboarding or instrumentation prompt copied by the CTA button. |
|  | `onCopy?` | `() => void` | Callback fired when the scan prompt is copied. |
|  | `tryFoglampHref?` | `string` | Destination href for the primary call-to-action link. |

## Usage

```tsx
import { ScanLanding } from "@tailrocks/framerock/blocks/scan/scan-landing";
import {
  ScanFaviconProvider,
} from "@tailrocks/framerock/blocks/scan/favicon-context";
import { getGoogleFavicon } from "@tailrocks/framerock/widgets/leaf/favicon";

const resolveStoryFavicon = getGoogleFavicon;

export function ScanLandingExample({
  tryFoglampHref = '#scan-demo',
}: {
  tryFoglampHref?: string;
} = {}) {
  return (
    <ScanFaviconProvider resolve={resolveStoryFavicon}>
      <div className="w-full">
        <ScanLanding tryFoglampHref={tryFoglampHref} onCopy={() => {}} />
      </div>
    </ScanFaviconProvider>
  );
}
```
