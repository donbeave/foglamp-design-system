## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `EmbedBoard` | `data` | `ScanData` | Structured dataset rendered by the block. |
|  | `slug` | `string` | Unique scan or resource slug identifier. |

## Usage

```tsx
import { EmbedBoard } from "@tailrocks/framerock/blocks/scan/embed-board";
import {
  ScanFaviconProvider,
} from "@tailrocks/framerock/blocks/scan/favicon-context";
import type { ScanData } from "@tailrocks/framerock/lib/scan/scan";
import { getGoogleFavicon } from "@tailrocks/framerock/widgets/leaf/favicon";

const scanFixture: ScanData = {
  version: 1,
  project: {
    name: "Framerock",
    slug: "framerock",
    tagline: "AI observability, clearly mapped",
    iconDomain: "tailrocks.com",
    date: "2026-11-05",
  },
  stats: { agents: 2, models: 2, tools: 2, integrations: 1 },
  topModels: [{ id: "gpt-4o", label: "GPT-4o", domain: "openai.com" }],
  topTools: [{ id: "search", label: "Search", domain: "exa.ai" }],
  topIntegrations: [{ id: "slack", label: "Slack", domain: "slack.com" }],
  graph: {
    nodes: [
      { id: "webhook", label: "Incoming webhook", kind: "entry" },
      { id: "support", label: "Support agent", kind: "agent" },
      { id: "openai", label: "GPT-4o", kind: "model", domain: "openai.com" },
    ],
    edges: [
      { from: "webhook", to: "support", kind: "triggers" },
      { from: "support", to: "openai", kind: "calls" },
    ],
  },
};

const resolveStoryFavicon = getGoogleFavicon;

export function EmbedBoardExample({
  slug = 'framerock',
}: {
  slug?: string;
} = {}) {
  return (
    <ScanFaviconProvider resolve={resolveStoryFavicon}>
      <div className="h-[720px] w-full overflow-hidden">
        <EmbedBoard data={scanFixture} slug={slug} />
      </div>
    </ScanFaviconProvider>
  );
}
```
