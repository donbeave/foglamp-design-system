## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `SetupBoardBlock` | `plan` | `DetectedPlan` | Detected instrumentation plan with agents, workflows, and sessions. |
|  | `approved?` | `Decisions \| null` | Finalized instrumentation decisions after review approval. |
|  | `applied?` | `AppliedReport \| null` | Agent completion report once instrumentation has been applied. |
|  | `status` | `PlanStatus` | Current lifecycle state of the setup review plan. |
|  | `firstTraceId?` | `string \| null` | Identifier of the first trace received after applying the plan. |
|  | `approvedAt?` | `string \| null` | ISO timestamp when the setup plan was approved. |
|  | `agentResumedAt?` | `string \| null` | ISO timestamp when the coding agent resumed execution. |
|  | `spanCount?` | `number \| null` | Number of spans captured in the verification trace. |
|  | `secondsToFirstTrace?` | `number \| null` | Elapsed seconds from plan approval to the first trace. |
|  | `failureStage?` | `string \| null` | Stage identifier when plan application fails. |
|  | `onApprove` | `(edits: PlanEdits \| undefined) => void` | Callback invoked with user edits when the setup plan is approved. |
|  | `onReject` | `() => void` | Callback invoked when the setup plan is rejected. |
|  | `approving` | `boolean` | Marks the approval action as in flight. |

## Usage

```tsx
import { useState } from "react";
import {
  SetupBoardBlock,
} from "@tailrocks/framerock/blocks/setup/setup-board";
import {
  ScanFaviconProvider,
} from "@tailrocks/framerock/blocks/scan/favicon-context";
import { PlanEdits } from "@tailrocks/framerock/lib/scan/instrumentation";
import type { DetectedPlan } from "@tailrocks/framerock/lib/scan/instrumentation";
import type { ScanData } from "@tailrocks/framerock/lib/scan/scan";
import { getGoogleFavicon } from "@tailrocks/framerock/widgets/leaf/favicon";

const scanFixture: ScanData = {
  version: 1,
  project: {
    name: "Framerock",
    slug: "framerock",
    tagline: "AI observability, clearly mapped",
    iconDomain: "tailrocks.com",
    date: "2026-11-05",
  },
  stats: { agents: 1, models: 1, tools: 1, integrations: 0 },
  topModels: [{ id: "gpt-4o", label: "GPT-4o", domain: "openai.com" }],
  topTools: [{ id: "search", label: "Search", domain: "exa.ai" }],
  topIntegrations: [],
  graph: {
    nodes: [
      { id: "support", label: "Support agent", kind: "agent" },
      { id: "openai", label: "GPT-4o", kind: "model", domain: "openai.com" },
    ],
    edges: [{ from: "support", to: "openai", kind: "calls" }],
  },
};

const planFixture: DetectedPlan = {
  version: 1,
  sdk: { major: 5, version: "5.0.12" },
  hasReactUi: true,
  scan: scanFixture,
  calls: [
    {
      id: "call-support",
      fn: "generateText",
      sourceRef: "src/agents/support.ts:42",
      modelId: "gpt-4o",
    },
  ],
  decisions: {
    agents: [
      {
        id: "support-agent",
        name: "Support agent",
        callIds: ["call-support"],
        oneOff: false,
        confidence: "high",
        sourceRef: "src/agents/support.ts:42",
        rationale: "Handles incoming customer support requests.",
      },
    ],
    workflows: [],
    sessions: [],
    customer: {
      recommended: false,
      confidence: "high",
      rationale: "Single-workspace deployment.",
    },
  },
};

const resolveStoryFavicon = getGoogleFavicon;

export function SetupPlanEditsExample({
  approving = false,
}: {
  approving?: boolean;
} = {}) {
  const [status, setStatus] = useState<'awaiting_approval' | 'approved' | 'rejected'>('awaiting_approval');
  const [, setEdits] = useState<PlanEdits | undefined>();
  return (
    <ScanFaviconProvider resolve={resolveStoryFavicon}>
      <SetupBoardBlock
        plan={planFixture}
        status={status}
        approving={approving}
        onApprove={edits => { setEdits(edits); setStatus('approved'); }}
        onReject={() => setStatus('rejected')}
      />
    </ScanFaviconProvider>
  );
}
```
