## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `FigureKit22Block` | `section?` | `Section` | Marketing figure category (`"costs"` \| `"traces"` \| `"quality"`). |
|  | `variantId?` | `string` | Figure variant identifier within the selected section. |

## Usage

```tsx
import { useState } from "react";
import {
  FigureKit22Block,
} from "@tailrocks/framerock/blocks/marketing/figures/figure-kit-22-block";
import {
  FIGURES,
} from "@tailrocks/framerock/blocks/marketing/figures/figures";

export function FigureKitExample({
  section = 'costs',
  variantId = 'chart',
}: {
  section?: 'costs' | 'traces' | 'quality';
  variantId?: string;
} = {}) {
  const [activeSection, setActiveSection] = useState<'costs' | 'traces' | 'quality'>(section);
  const [activeVariant, setActiveVariant] = useState<string>(variantId);
  return (
    <FigureKit22Block
      key={`${activeSection}-${activeVariant}`}
      section={activeSection}
      variantId={activeVariant}
    />
  );
}
```
