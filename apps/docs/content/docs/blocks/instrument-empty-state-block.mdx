## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `InstrumentEmptyStateBlock` | `feature` | `InstrumentFeature` | Target observability feature (`"agent"` \| `"workflow"` \| `"session"`). |
|  | `icon` | `React.ComponentType<{ className?: string }>` | Header icon component rendered inside the empty state card. |
|  | `title` | `string` | Primary heading text. |
|  | `description?` | `string` | Supporting description text. |

## Usage

```tsx
import {
  InstrumentEmptyStateBlock,
} from "@tailrocks/framerock/blocks/auth/instrument-empty-state";
import {
  BrandMark,
} from "@tailrocks/framerock/blocks/marketing/chrome/brand-mark";

export function InstrumentEmptyStateExample({
  feature = 'agent',
  title = 'No agent traces yet',
}: {
  feature?: 'agent' | 'workflow' | 'session';
  title?: string;
} = {}) {
  return (
    <InstrumentEmptyStateBlock
      feature={feature}
      title={title}
      icon={BrandMark}
      description="Add observability to your first AI workflow."
    />
  );
}
```
