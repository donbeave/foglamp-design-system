## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `LoginFormBlock` | `methods?` | `AuthMethods` | Enabled sign-in methods (`emailPassword`, `magicLink`, `google`). |
|  | `next?` | `string \| null` | Sanitized post-authentication redirect path. |
|  | `resetPasswordHref?` | `string` | Route path for the host password reset screen. |
|  | `onGoogleSignIn?` | `(input: { callbackURL: string; }) => Promise<LoginFormActionResult>` | Async handler for Google OAuth sign-in. |
|  | `onPasswordSignIn?` | `(input: { email: string; password: string; callbackURL: string; }) => Promise<LoginFormActionResult>` | Async handler for email and password sign-in. |
|  | `onPasswordReset?` | `(input: { email: string; redirectTo: string; }) => Promise<LoginFormActionResult>` | Async handler for password reset requests. |
|  | `onMagicLinkSignIn?` | `(input: { email: string; callbackURL: string; }) => Promise<LoginFormActionResult>` | Async handler for passwordless magic-link sign-in. |
|  | `onSignedIn?` | `(completion: LoginFormCompletion) => void` | Callback invoked after authentication completes. |

## Usage

```tsx
import { LoginFormBlock } from "@tailrocks/framerock/blocks/auth/login-form";
import {
  LoginFormCompletion,
} from "@tailrocks/framerock/blocks/auth/login-form";
import { useState } from "react";

export function LoginFormExample({
  magicLink = true,
}: {
  magicLink?: boolean;
} = {}) {
  const [done, setDone] = useState<LoginFormCompletion | null>(null);
  return (
    <div className="mx-auto flex w-full max-w-md justify-center">
      {done ? (
          <p role="status" className="rounded-xl bg-muted p-4">
            Local demo signed in with
            {done.method}
            .
          </p>
      ) : (
          <LoginFormBlock
            methods={{ emailPassword: true, magicLink, google: true }}
            onSignedIn={setDone}
          />
      )}
    </div>
  );
}
```
