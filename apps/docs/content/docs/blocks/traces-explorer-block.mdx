## Public API

| Export | Prop | Type | Description |
| :--- | :--- | :--- | :--- |
| `TracesExplorerBlock` | `traces?` | `TraceExplorerRow[]` | Map of trace IDs to span arrays for the trace waterfall preview. |
|  | `isLoading?` | `boolean` | Renders loading skeleton placeholders while data is being fetched. |
|  | `isFetching?` | `boolean` | Indicates background pagination or filter refetching. |
|  | `promptVersions?` | `{ id: string; agentName: string; number: number; current: boolean }[]` | Available prompt versions for filtering traces. |
|  | `onTraceSelect?` | `(traceId: string) => void` | Callback fired when a trace row is selected. |
|  | `onFiltersChange?` | `(filters: TracesExplorerFilters) => void` | Callback fired when search, filter, or sort controls change. |

## Usage

```tsx
import { useState } from "react";
import {
  TracesExplorerBlock,
} from "@tailrocks/framerock/blocks/dashboard/traces-explorer-block";
import {
  TraceWaterfallInspectorBlock,
} from "@tailrocks/framerock/blocks/dashboard/trace-waterfall-inspector-block";

export function TracesExplorerExample() {
  const [traceId, setTraceId] = useState<string | null>(null);

  if (traceId) {
    return (
      <div>
        <button
          type="button"
          className="mb-3 text-sm text-muted-foreground hover:text-foreground"
          onClick={() => setTraceId(null)}
        >
          ← Back to traces
        </button>
        <TraceWaterfallInspectorBlock traceId={traceId} />
      </div>
    );
  }

  return (
    <TracesExplorerBlock
      onTraceSelect={setTraceId}
      onFiltersChange={() => {}}
    />
  );
}
```
