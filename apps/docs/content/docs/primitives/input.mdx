## Usage

```tsx
import { Input } from "@tailrocks/framerock/ui/input";
import { Label } from "@tailrocks/framerock/ui/label";

export function InputExample({
  type = 'text',
  placeholder = 'Project name',
}: {
  type?: 'text' | 'email' | 'search';
  placeholder?: string;
} = {}) {
  return (
    <div className="grid w-full max-w-sm gap-2">
      <Label htmlFor="input-demo">
        {type === 'email' ? 'Notification email' : type === 'search' ? 'Search traces' : 'Project name'}
      </Label>
      <Input
        id="input-demo"
        type={type}
        placeholder={placeholder}
        defaultValue={type === 'text' ? 'Support Agent' : undefined}
      />
    </div>
  );
}
```
