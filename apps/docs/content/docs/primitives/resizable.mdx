## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `ResizableHandle` | `ResizablePrimitive.SeparatorProps & { withHandle?: boolean; }` | Panel resize handle with an optional visual grip. |
| `ResizablePanel` | `ResizablePrimitive.PanelProps` | Resizable panel region. |
| `ResizablePanelGroup` | `ResizablePrimitive.GroupProps` | Resizable panel group root. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `ResizableHandle.withHandle` | `boolean` | Render the visual grip. |

## Usage

```tsx
import {
  ResizablePanelGroup,
  ResizablePanel,
  ResizableHandle,
} from "@tailrocks/framerock/ui/resizable";

export function ResizableExample({
  direction = "horizontal",
  withHandle = true,
}: {
  direction?: "horizontal" | "vertical";
  withHandle?: boolean;
} = {}) {
  return (
    <ResizablePanelGroup
      orientation={direction}
      className="min-h-48 w-full max-w-xl rounded-lg shadow-(--custom-shadow)"
    >
      <ResizablePanel defaultSize={36} minSize={25}>
        <div className="h-full p-4">
          <p
            className="mb-3 text-xs font-medium uppercase text-muted-foreground"
          >
            Trace list
          </p>
          <div className="rounded-md bg-muted p-3 text-sm">
            POST /checkout
          </div>
        </div>
      </ResizablePanel>
      <ResizableHandle withHandle={withHandle} />
      <ResizablePanel defaultSize={64}>
        <div className="h-full p-4">
          <p
            className="mb-3 text-xs font-medium uppercase text-muted-foreground"
          >
            Span details
          </p>
          <p className="text-sm">payment.authorize</p>
          <p className="mt-2 text-xs text-muted-foreground">
            Duration · 182 ms
          </p>
        </div>
      </ResizablePanel>
    </ResizablePanelGroup>
  );
}
```
