## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `ToggleGroup` | `ToggleGroupPrimitive.Props & VariantProps<typeof toggleVariants> & { spacing?: number; orientation?: "horizontal" \| "vertical"; }` | Toggle group with shared variant, size, orientation, and spacing. |
| `ToggleGroupItem` | `TogglePrimitive.Props & VariantProps<typeof toggleVariants>` | Toggle button that shares the group state and options. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `ToggleGroup.variant` | `Toggle variant` | Shared toggle treatment. |
| `ToggleGroup.size` | `Toggle size` | Shared toggle dimensions. |
| `ToggleGroup.spacing` | `number` | Defaults to `0`. |
| `ToggleGroup.orientation` | `horizontal \| vertical` | Defaults to `horizontal`. |

## Usage

```tsx
import { useState } from "react";
import {
  ToggleGroup,
  ToggleGroupItem,
} from "@tailrocks/framerock/ui/toggle-group";

export function ToggleGroupExample({
  defaultValue = "24h",
  variant = "outline",
}: {
  defaultValue?: "1h" | "24h" | "7d";
  variant?: "default" | "outline";
} = {}) {
  const [period, setPeriod] = useState<string>(defaultValue);
  return (
    <div className="grid gap-3">
      <ToggleGroup
        variant={variant}
        value={[period]}
        onValueChange={(next) => { const v = Array.isArray(next) ? next[0] : next; if (typeof v === "string" && v) setPeriod(v); }}
        aria-label="Time period"
      >
        <ToggleGroupItem value="1h" aria-label="Last hour">
          1h
        </ToggleGroupItem>
        <ToggleGroupItem value="24h" aria-label="Last day">
          24h
        </ToggleGroupItem>
        <ToggleGroupItem value="7d" aria-label="Last week">
          7d
        </ToggleGroupItem>
      </ToggleGroup>
      <p className="text-xs text-muted-foreground">
        Showing activity for
        {period}
        .
      </p>
    </div>
  );
}
```
