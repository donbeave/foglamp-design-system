## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `AspectRatio` | `React.ComponentProps<"div"> & { ratio: number }` | Container constrained to a numeric aspect ratio. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `AspectRatio.ratio` | `number` | Required width-to-height ratio. |

## Usage

```tsx
import { AspectRatio } from "@tailrocks/framerock/ui/aspect-ratio";

export function AspectRatioExample({
  ratio = 16 / 9,
}: {
  ratio?: number;
} = {}) {
  return (
    <div className="w-full max-w-xl">
      <AspectRatio
        ratio={ratio}
        className="overflow-hidden rounded-lg bg-muted"
      >
        <div
          className="flex size-full flex-col items-center justify-center gap-2 bg-gradient-to-br from-primary/15 via-muted to-primary/5"
        >
          <span className="text-2xl font-semibold">Trace overview</span>
          <span className="text-sm text-muted-foreground">
            {ratio.toFixed(2)}
            : 1 preview
          </span>
        </div>
      </AspectRatio>
    </div>
  );
}
```
