## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Sidebar` | `React.ComponentProps<"div"> & { side?: "left" \| "right"; variant?: "sidebar" \| "floating" \| "inset"; collapsible?: "offcanvas" \| "icon" \| "none"; }` | Sidebar layout container. |
| `SidebarContent` | `React.ComponentProps<"div">` | Contains the associated popup or panel content. |
| `SidebarFooter` | `React.ComponentProps<"div">` | Footer slot for actions or supporting content. |
| `SidebarGroup` | `React.ComponentProps<"div">` | Groups related entries. |
| `SidebarGroupAction` | `useRender.ComponentProps<"button"> & React.ComponentProps<"button">` | Action control aligned with a sidebar group label. |
| `SidebarGroupContent` | `React.ComponentProps<"div">` | Content area inside a sidebar group. |
| `SidebarGroupLabel` | `useRender.ComponentProps<"div"> & React.ComponentProps<"div">` | Label for the associated control or group. |
| `SidebarHeader` | `React.ComponentProps<"div">` | Header slot for title and supporting content. |
| `SidebarInput` | `React.ComponentProps<typeof Input>` | Input control for the composition. |
| `SidebarInset` | `React.ComponentProps<"main">` | Main page area positioned beside the sidebar. |
| `SidebarMenu` | `React.ComponentProps<"ul">` | List container for sidebar menu rows. |
| `SidebarMenuAction` | `useRender.ComponentProps<"button"> & React.ComponentProps<"button"> & { showOnHover?: boolean; }` | Action control aligned with a sidebar menu row. |
| `SidebarMenuBadge` | `React.ComponentProps<"div">` | Badge aligned with a sidebar menu row. |
| `SidebarMenuButton` | `useRender.ComponentProps<"button"> & React.ComponentProps<"button"> & { isActive?: boolean; tooltip?: string \| React.ComponentProps<typeof TooltipContent>; } & VariantProps<typeof sidebarMenuButtonVariants>` | Sidebar menu link or button with active, variant, size, and tooltip options. |
| `SidebarMenuItem` | `React.ComponentProps<"li">` | List item that contains a sidebar menu row. |
| `SidebarMenuSkeleton` | `React.ComponentProps<"div"> & { showIcon?: boolean; }` | Placeholder for a sidebar menu row. |
| `SidebarMenuSub` | `React.ComponentProps<"ul">` | Nested sidebar menu list. |
| `SidebarMenuSubButton` | `useRender.ComponentProps<"a"> & React.ComponentProps<"a"> & { size?: "sm" \| "md"; isActive?: boolean; }` | Link-style control in a nested sidebar menu. |
| `SidebarMenuSubItem` | `React.ComponentProps<"li">` | List item in a nested sidebar menu. |
| `SidebarProvider` | `React.ComponentProps<"div"> & { defaultOpen?: boolean; open?: boolean; onOpenChange?: (open: boolean) => void; }` | Provides shared context to related components. |
| `SidebarRail` | `React.ComponentProps<"button">` | Narrow sidebar interaction rail. |
| `SidebarSeparator` | `React.ComponentProps<typeof Separator>` | Visual or semantic divider. |
| `SidebarTrigger` | `React.ComponentProps<typeof Button>` | Opens or toggles the associated popup or panel. |
| `useSidebar` | `Exported component or value` | Reads the sidebar state and controls. |

| `Sidebar.variant` | `sidebar \| floating \| inset` | Defaults to `sidebar`. |

## Usage

```tsx
import {
  SidebarProvider,
  Sidebar,
  SidebarHeader,
  SidebarContent,
  SidebarGroup,
  SidebarGroupLabel,
  SidebarMenu,
  SidebarMenuItem,
  SidebarMenuButton,
  SidebarInset,
  SidebarTrigger,
} from "@tailrocks/framerock/ui/sidebar";

export function SidebarExample({
  collapsible = "icon",
  activeItem = "Traces",
}: {
  collapsible?: "offcanvas" | "icon" | "none";
  activeItem?: "Overview" | "Traces" | "Evaluations";
} = {}) {
  const navItems: Array<"Overview" | "Traces" | "Evaluations"> = ["Overview", "Traces", "Evaluations"];
  return (
    <div
      className="w-full overflow-hidden rounded-lg border border-border/60"
    >
      <SidebarProvider defaultOpen>
        <Sidebar collapsible={collapsible}>
          <SidebarHeader className="p-3 font-semibold">
            Framerock
          </SidebarHeader>
          <SidebarContent>
            <SidebarGroup>
              <SidebarGroupLabel>Workspace</SidebarGroupLabel>
              <SidebarMenu>
                {navItems.map((item) => (
                    <SidebarMenuItem key={item}>
                      <SidebarMenuButton isActive={activeItem === item}>
                        {item}
                      </SidebarMenuButton>
                    </SidebarMenuItem>
                ))}
              </SidebarMenu>
            </SidebarGroup>
          </SidebarContent>
        </Sidebar>
        <SidebarInset className="min-h-64">
          <header className="flex items-center gap-3 border-b p-3">
            <SidebarTrigger />
            <span className="text-sm font-medium">
              {activeItem}
            </span>
          </header>
          <div className="p-4 text-sm text-muted-foreground">
            Workspace navigation with responsive collapse behavior.
          </div>
        </SidebarInset>
      </SidebarProvider>
    </div>
  );
}
```
