## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Card` | `React.ComponentProps<"div"> & { size?: "default" \| "sm" }` | Card container with default or compact density. |
| `CardAction` | `React.ComponentProps<"div">` | Action slot or action control. |
| `CardContent` | `React.ComponentProps<"div">` | Contains the associated popup or panel content. |
| `CardDescription` | `React.ComponentProps<"div">` | Descriptive text for the associated content. |
| `CardFooter` | `React.ComponentProps<"div">` | Footer slot for actions or supporting content. |
| `CardHeader` | `React.ComponentProps<"div">` | Header slot for title and supporting content. |
| `CardTitle` | `React.ComponentProps<"div">` | Title text for the associated content. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `Card.size` | `default \| sm` | Defaults to `default`. |

## Usage

```tsx
import {
  Card,
  CardHeader,
  CardTitle,
  CardDescription,
  CardAction,
  CardContent,
  CardFooter,
} from "@tailrocks/framerock/ui/card";
import { Button } from "@tailrocks/framerock/ui/button";
import { Badge } from "@tailrocks/framerock/ui/badge";

export function CardExample({
  size = 'default',
}: {
  size?: 'default' | 'sm';
} = {}) {
  return (
    <Card size={size} className="w-full max-w-md">
      <CardHeader>
        <CardTitle>Production evaluation</CardTitle>
        <CardDescription>
          Customer support agent · updated 4 min ago
        </CardDescription>
        <CardAction>
          <Badge variant="green">Passed</Badge>
        </CardAction>
      </CardHeader>
      <CardContent>
        <div className="flex items-baseline gap-2">
          <span className="text-3xl font-semibold">94.8</span>
          <span className="text-muted-foreground">/ 100</span>
        </div>
        <p className="mt-1 text-sm text-muted-foreground">
          Across 128 test cases
        </p>
      </CardContent>
      <CardFooter className="justify-between border-t">
        <span className="text-xs text-muted-foreground">v2.8.1 · 2m 14s</span>
        <Button size="sm" variant="outline">Open report</Button>
      </CardFooter>
    </Card>
  );
}
```
