## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Table` | `React.ComponentProps<"table"> & {  maxHeight?: number \| string;  stickyHeader?: boolean; }` | Semantic table with optional sticky header and internal scrolling. |
| `TableBody` | `React.ComponentProps<"tbody">` | Table body section containing data rows. |
| `TableCaption` | `React.ComponentProps<"caption">` | Caption that labels the table. |
| `TableCell` | `Omit<React.ComponentProps<"td">, "align"> & { align?: Align }` | Data cell with start, center, or end alignment. |
| `TableFooter` | `React.ComponentProps<"tfoot">` | Table footer section for summary or totals. |
| `TableHead` | `Omit<React.ComponentProps<"th">, "align"> & { align?: Align }` | Header cell with start, center, or end alignment. |
| `TableHeader` | `React.ComponentProps<"thead">` | Table head section containing header rows. |
| `TableRow` | `React.ComponentProps<"tr"> & {  interactive?: boolean; }` | Semantic row. Set interactive to enable the component’s keyboard and hover behavior. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `Table.maxHeight` | `number \| string` | Optional cap; the body scrolls internally when set. |
| `Table.stickyHeader` | `boolean` | Pins the header. Auto-enabled when `maxHeight` is set. |
| `TableRow.interactive` | `boolean` | Enables pointer and hover treatment plus Enter and Space activation. |
| `TableHead.align` | `start \| center \| end` | Header content alignment. |
| `TableCell.align` | `start \| center \| end` | Cell content alignment. |

## Usage

```tsx
import { useState } from "react";
import {
  Table,
  TableHeader,
  TableBody,
  TableFooter,
  TableHead,
  TableRow,
  TableCell,
  TableCaption,
} from "@tailrocks/framerock/ui/table";

export function TableExample({
  showFooter = true,
  dense = false,
}: {
  showFooter?: boolean;
  dense?: boolean;
} = {}) {
  const [selected, setSelected] = useState("tr_002");
  const rows = [
    { id: "tr_001", model: "gpt-4.1", latency: 421 },
    { id: "tr_002", model: "claude-sonnet", latency: 876 },
    { id: "tr_003", model: "o4-mini", latency: 193 },
  ];
  const cellClass = dense ? "py-1.5 text-xs" : undefined;
  return (
    <div
      className="max-h-72 w-full max-w-lg overflow-auto rounded-md shadow-(--custom-shadow)"
    >
      <Table>
        <TableCaption>Recent model requests</TableCaption>
        <TableHeader className="sticky top-0 bg-background">
          <TableRow>
            <TableHead>Request</TableHead>
            <TableHead>Model</TableHead>
            <TableHead className="text-right">Latency</TableHead>
          </TableRow>
        </TableHeader>
        <TableBody>
          {rows.map((row) => (
              <TableRow
                key={row.id}
                onClick={() => setSelected(row.id)}
                data-state={selected === row.id ? "selected" : undefined}
                className="cursor-pointer"
              >
                <TableCell className={cellClass}>
                  {row.id}
                </TableCell>
                <TableCell className={cellClass}>
                  {row.model}
                </TableCell>
                <TableCell className={`text-right ${cellClass ?? ""}`}>
                  {row.latency}
                  ms
                </TableCell>
              </TableRow>
          ))}
        </TableBody>
        {showFooter ? (
            <TableFooter>
              <TableRow>
                <TableCell colSpan={2}>Average latency</TableCell>
                <TableCell className="text-right">497 ms</TableCell>
              </TableRow>
            </TableFooter>
        ) : null}
      </Table>
    </div>
  );
}
```
