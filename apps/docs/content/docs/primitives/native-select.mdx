## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `NativeSelect` | `NativeSelectProps` | Styled native select element. |
| `NativeSelectOptGroup` | `React.ComponentProps<"optgroup">` | Native option group element. |
| `NativeSelectOption` | `React.ComponentProps<"option">` | Native option element. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `NativeSelect.size` | `default \| sm` | Defaults to `default`. |

## Usage

```tsx
import {
  NativeSelect,
  NativeSelectOption,
} from "@tailrocks/framerock/ui/native-select";
import { Label } from "@tailrocks/framerock/ui/label";

export function NativeSelectExample({
  defaultRegion = "us-east",
  size = "default",
  disabled = false,
}: {
  defaultRegion?: "us-east" | "eu-west" | "ap-south";
  size?: "sm" | "default";
  disabled?: boolean;
} = {}) {
  return (
    <div className="grid w-full max-w-xs gap-2">
      <Label htmlFor="region">Deployment region</Label>
      <NativeSelect
        key={defaultRegion}
        id="region"
        size={size}
        disabled={disabled}
        defaultValue={defaultRegion}
      >
        <NativeSelectOption value="us-east">US East</NativeSelectOption>
        <NativeSelectOption value="eu-west">EU West</NativeSelectOption>
        <NativeSelectOption value="ap-south">Asia Pacific</NativeSelectOption>
      </NativeSelect>
      <p className="text-xs text-muted-foreground">
        Uses the browser’s native selection menu.
      </p>
    </div>
  );
}
```
