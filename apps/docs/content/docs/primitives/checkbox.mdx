## Usage

```tsx
import { useState } from "react";
import { Checkbox } from "@tailrocks/framerock/ui/checkbox";
import {
  Field,
  FieldLabel,
  FieldDescription,
} from "@tailrocks/framerock/ui/field";

export function CheckboxExample({
  disabled = false,
}: {
  disabled?: boolean;
} = {}) {
  const [checked, setChecked] = useState(true);
  return (
    <Field orientation="horizontal" data-disabled={disabled}>
      <Checkbox
        id="include-retries"
        checked={checked}
        onCheckedChange={setChecked}
        disabled={disabled}
      />
      <div>
        <FieldLabel htmlFor="include-retries">
          Include retry attempts
        </FieldLabel>
        <FieldDescription>
          Count retries as separate evaluation requests.
        </FieldDescription>
      </div>
    </Field>
  );
}
```
