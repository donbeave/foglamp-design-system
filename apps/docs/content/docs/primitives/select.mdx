## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Select` | `SelectPrimitive.Root.Props<Value, Multiple>` | Selection root that collects nested labels when items are omitted. |
| `SelectContent` | `SelectPrimitive.Popup.Props & Pick< SelectPrimitive.Positioner.Props, "align" \| "alignOffset" \| "side" \| "sideOffset" \| "alignItemWithTrigger" >` | Positioned select popup. |
| `SelectGroup` | `SelectPrimitive.Group.Props` | Groups related entries. |
| `SelectItem` | `SelectPrimitive.Item.Props` | One selectable option in the select list. |
| `SelectLabel` | `SelectPrimitive.GroupLabel.Props` | Label for the associated control or group. |
| `SelectScrollDownButton` | `React.ComponentProps<typeof SelectPrimitive.ScrollDownArrow>` | Control that scrolls the select popup toward later items. |
| `SelectScrollUpButton` | `React.ComponentProps<typeof SelectPrimitive.ScrollUpArrow>` | Control that scrolls the select popup toward earlier items. |
| `SelectSeparator` | `SelectPrimitive.Separator.Props` | Visual or semantic divider. |
| `SelectTrigger` | `SelectPrimitive.Trigger.Props & { size?: "sm" \| "default"; }` | Control that opens the select popup. |
| `SelectValue` | `SelectPrimitive.Value.Props` | Displays the selected select value. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `Select.items` | `Base UI select items` | Optional item mapping or list. If omitted, the root collects `SelectItem` labels from nested children. |
| `SelectTrigger.size` | `sm \| default` | Defaults to `default`. |
| `SelectContent.side` | `Base UI positioner side` | Defaults to `bottom`. |
| `SelectContent.sideOffset` | `number` | Defaults to `4`. |
| `SelectContent.align` | `Base UI positioner alignment` | Defaults to `center`. |
| `SelectContent.alignOffset` | `number` | Defaults to `0`. |
| `SelectContent.alignItemWithTrigger` | `boolean` | Defaults to `false`. |
| `SelectItem.label` | `React.ReactNode` | Explicit item label used by child-label collection. |

## Usage

```tsx
import { useState } from "react";
import {
  Select,
  SelectTrigger,
  SelectValue,
  SelectContent,
  SelectGroup,
  SelectLabel,
  SelectItem,
  SelectSeparator,
} from "@tailrocks/framerock/ui/select";
import { Label } from "@tailrocks/framerock/ui/label";

export function SelectExample({
  placeholder = "Select model",
  disabled = false,
}: {
  placeholder?: string;
  disabled?: boolean;
} = {}) {
  const [model, setModel] = useState("gpt-4.1");
  return (
    <div className="grid w-full max-w-xs gap-2">
      <Label>Model</Label>
      <Select
        value={model}
        disabled={disabled}
        onValueChange={(value) => value && setModel(value)}
      >
        <SelectTrigger>
          <SelectValue placeholder={placeholder} />
        </SelectTrigger>
        <SelectContent>
          <SelectGroup>
            <SelectLabel>OpenAI</SelectLabel>
            <SelectItem value="gpt-4.1">GPT-4.1</SelectItem>
            <SelectItem value="o4-mini">o4-mini</SelectItem>
          </SelectGroup>
          <SelectSeparator />
          <SelectGroup>
            <SelectLabel>Anthropic</SelectLabel>
            <SelectItem value="claude-sonnet">Claude Sonnet</SelectItem>
          </SelectGroup>
        </SelectContent>
      </Select>
      <p className="text-xs text-muted-foreground">
        Selected:
        {model}
      </p>
    </div>
  );
}
```
