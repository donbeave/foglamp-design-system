## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Tooltip` | `TooltipPrimitive.Root.Props` | Tooltip state root. |
| `TooltipContent` | `TooltipPrimitive.Popup.Props & Pick< TooltipPrimitive.Positioner.Props, "align" \| "alignOffset" \| "side" \| "sideOffset" >` | Positioned tooltip popup. |
| `TooltipProvider` | `TooltipPrimitive.Provider.Props` | Provider for tooltip delay and shared state. |
| `TooltipTrigger` | `TooltipPrimitive.Trigger.Props` | Element that triggers the tooltip. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `TooltipProvider.delay` | `number` | Open delay in milliseconds. Defaults to `0`. |
| `TooltipContent.side` | `Base UI positioner side` | Defaults to `top`. |
| `TooltipContent.sideOffset` | `number` | Defaults to `4`. |
| `TooltipContent.align` | `Base UI positioner alignment` | Defaults to `center`. |
| `TooltipContent.alignOffset` | `number` | Defaults to `0`. |

## Usage

```tsx
import {
  TooltipProvider,
  Tooltip,
  TooltipTrigger,
  TooltipContent,
} from "@tailrocks/framerock/ui/tooltip";

export function TooltipExample({
  content = "Percentage of requests captured for analysis.",
  side = "top",
}: {
  content?: string;
  side?: "top" | "right" | "bottom" | "left";
} = {}) {
  return (
    <TooltipProvider>
      <div className="flex items-center gap-2">
        <span className="text-sm">Sampling rate</span>
        <Tooltip>
          <TooltipTrigger
            className="inline-flex size-6 items-center justify-center rounded-full border border-border/60 text-xs"
            aria-label="About sampling"
          >
            ⓘ
          </TooltipTrigger>
          <TooltipContent side={side}>
            {content}
          </TooltipContent>
        </Tooltip>
      </div>
    </TooltipProvider>
  );
}
```
