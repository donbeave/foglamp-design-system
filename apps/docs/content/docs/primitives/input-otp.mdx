## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `InputOTP` | `React.ComponentProps<typeof OTPInput> & { containerClassName?: string; }` | OTP input root. |
| `InputOTPGroup` | `React.ComponentProps<"div">` | Groups related entries. |
| `InputOTPSeparator` | `React.ComponentProps<"div">` | Visual or semantic divider. |
| `InputOTPSlot` | `React.ComponentProps<"div"> & { index: number; }` | Cell at the required OTP input index. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `InputOTP.containerClassName` | `string` | Class for the OTP package container. |
| `InputOTPSlot.index` | `number` | Required slot index. |

## Usage

```tsx
import { useState } from "react";
import {
  InputOTP,
  InputOTPGroup,
  InputOTPSlot,
  InputOTPSeparator,
} from "@tailrocks/framerock/ui/input-otp";

export function InputOTPExample({
  maxLength = 6,
}: {
  maxLength?: number;
} = {}) {
  const [value, setValue] = useState('482');
  return (
    <div className="grid gap-3">
      <p className="text-sm font-medium">Enter verification code</p>
      <InputOTP
        maxLength={maxLength}
        value={value}
        onChange={setValue}
        aria-label="Verification code"
      >
        <InputOTPGroup>
          {Array.from({ length: Math.min(3, maxLength) }, (_, index) => (
              <InputOTPSlot key={index} index={index} />
          ))}
        </InputOTPGroup>
        <InputOTPSeparator />
        <InputOTPGroup>
          {Array.from({ length: Math.max(0, maxLength - 3) }, (_, index) => (
              <InputOTPSlot key={index + 3} index={index + 3} />
          ))}
        </InputOTPGroup>
      </InputOTP>
      <p className="text-xs text-muted-foreground">
        Sent to your workspace administrator.
      </p>
    </div>
  );
}
```
