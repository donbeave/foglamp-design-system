## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `DirectionProvider` | `React.FC<DirectionProvider.Props>` | Provides direction context to Base UI components. |
| `useDirection` | `() => TextDirection` | Reads the direction context. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `DirectionProvider.direction` | `TextDirection` | Direction provided to descendant components. Defaults to `ltr`. |

## Usage

```tsx
import {
  DirectionProvider,
  useDirection,
} from "@tailrocks/framerock/ui/direction";
import { Button } from "@tailrocks/framerock/ui/button";

function DirectionContent() {
  const direction = useDirection();
  return (
    <div dir={direction} className="grid gap-3">
      <p className="text-sm">
        Current direction:
        <strong>
          {direction}
        </strong>
      </p>
      <div className="flex gap-2">
        <Button variant="outline">Previous</Button>
        <Button>Continue</Button>
      </div>
    </div>
  );
}

export function DirectionExample({
  direction = 'ltr',
}: {
  direction?: 'ltr' | 'rtl';
} = {}) {
  return (
    <DirectionProvider direction={direction}>
      <DirectionContent />
    </DirectionProvider>
  );
}
```
