## Usage

```tsx
import { useState } from "react";
import {
  RadioGroup,
  RadioGroupItem,
} from "@tailrocks/framerock/ui/radio-group";

export function RadioGroupExample({
  defaultValue = "balanced",
  disabled = false,
}: {
  defaultValue?: "fast" | "balanced" | "thorough";
  disabled?: boolean;
} = {}) {
  const [value, setValue] = useState<string>(defaultValue);
  const options = [
    ["fast", "Fast · low latency"],
    ["balanced", "Balanced · default judge"],
    ["thorough", "Thorough · multi-step rubric"],
  ] as const;
  return (
    <fieldset className="grid gap-3">
      <legend className="mb-2 text-sm font-medium">Evaluation mode</legend>
      <RadioGroup
        value={value}
        onValueChange={(next) => { if (typeof next === "string") setValue(next); }}
        disabled={disabled}
      >
        {options.map(([optionValue, label]) => (
            <label
              key={optionValue}
              className="flex cursor-pointer items-center gap-3"
            >
              <RadioGroupItem value={optionValue} />
              <span className="text-sm">
                {label}
              </span>
            </label>
        ))}
      </RadioGroup>
    </fieldset>
  );
}
```
