## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Slider` | `SliderPrimitive.Root.Props` | Slider input root. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `Slider.min` | `number` | Defaults to `0`. |
| `Slider.max` | `number` | Defaults to `100`. |
| `Slider.defaultValue` | `Base UI slider value` | Initial uncontrolled value. |
| `Slider.value` | `Base UI slider value` | Controlled value. |

## Usage

```tsx
import { useState } from "react";
import { Slider } from "@tailrocks/framerock/ui/slider";
import { Label } from "@tailrocks/framerock/ui/label";

export function SliderExample({
  defaultValue = 65,
  step = 5,
  disabled = false,
}: {
  defaultValue?: number;
  step?: number;
  disabled?: boolean;
} = {}) {
  const [value, setValue] = useState(defaultValue);
  return (
    <div className="grid w-full max-w-sm gap-4">
      <div className="flex justify-between text-sm">
        <Label>Sampling rate</Label>
        <span>
          {value}
          %
        </span>
      </div>
      <Slider
        value={[value]}
        min={0}
        max={100}
        step={step}
        disabled={disabled}
        onValueChange={(v) => setValue(typeof v === "number" ? v : (v[0] ?? 0))}
        aria-label="Sampling rate"
      />
      <p className="text-xs text-muted-foreground">
        Adjust the percentage of requests recorded.
      </p>
    </div>
  );
}
```
