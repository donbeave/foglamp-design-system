## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Tabs` | `TabsPrimitive.Root.Props` | Tabs state root. |
| `TabsContent` | `TabsPrimitive.Panel.Props` | Contains the associated popup or panel content. |
| `TabsList` | `TabsPrimitive.List.Props & VariantProps<typeof tabsListVariants>` | List container for related entries. |
| `tabsListVariants` | `Class variance recipe` | Exports the tabs class recipe. |
| `TabsTrigger` | `TabsPrimitive.Tab.Props` | Opens or toggles the associated popup or panel. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `Tabs.orientation` | `horizontal \| vertical` | Defaults to `horizontal`. |
| `TabsList.variant` | `default \| line` | Defaults to `default`. |

## Usage

```tsx
import { useState } from "react";
import {
  Tabs,
  TabsList,
  TabsTrigger,
  TabsContent,
} from "@tailrocks/framerock/ui/tabs";

export function TabsExample({
  defaultTab = "overview",
  showBadges = true,
}: {
  defaultTab?: "overview" | "spans" | "evals";
  showBadges?: boolean;
} = {}) {
  const [tab, setTab] = useState<string>(defaultTab);
  return (
    <Tabs
      value={tab}
      onValueChange={(next) => setTab(String(next))}
      className="w-full max-w-md"
    >
      <TabsList>
        <TabsTrigger value="overview">Overview</TabsTrigger>
        <TabsTrigger value="spans">
          Spans
          {showBadges ? "(12)" : ""}
        </TabsTrigger>
        <TabsTrigger value="evals">
          Evals
          {showBadges ? "(3)" : ""}
        </TabsTrigger>
      </TabsList>
      <TabsContent
        value="overview"
        className="rounded-md p-4 text-sm shadow-(--custom-shadow)"
      >
        Request completed in 1.42 seconds across 12 spans.
      </TabsContent>
      <TabsContent
        value="spans"
        className="rounded-md p-4 font-mono text-xs shadow-(--custom-shadow)"
      >
        chat.completions · 840 ms · 1,284 input tokens
      </TabsContent>
      <TabsContent
        value="evals"
        className="rounded-md p-4 text-sm shadow-(--custom-shadow)"
      >
        Faithfulness 0.96 · Answer relevance 0.94
      </TabsContent>
    </Tabs>
  );
}
```
