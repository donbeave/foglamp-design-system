## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Toggle` | `TogglePrimitive.Props & VariantProps<typeof toggleVariants>` | Toggle button with variant and size options. |
| `toggleVariants` | `Class variance recipe` | Exports the toggle class recipe. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `Toggle.variant` | `default \| outline` | Defaults to `default`. |
| `Toggle.size` | `default \| sm \| lg` | Defaults to `default`. |

## Usage

```tsx
import { useState } from "react";
import { Toggle } from "@tailrocks/framerock/ui/toggle";

export function ToggleExample({
  label = "Pin trace",
  defaultPressed = true,
  disabled = false,
}: {
  label?: string;
  defaultPressed?: boolean;
  disabled?: boolean;
} = {}) {
  const [pressed, setPressed] = useState(defaultPressed);
  return (
    <div className="grid gap-3">
      <div className="flex items-center gap-2">
        <Toggle
          aria-label={label}
          pressed={pressed}
          disabled={disabled}
          onPressedChange={setPressed}
        >
          {label}
        </Toggle>
        <Toggle
          aria-label="Highlight errors"
          variant="outline"
          disabled={disabled}
        >
          Errors only
        </Toggle>
      </div>
      <p className="text-xs text-muted-foreground">
        State:
        {pressed ? "Pinned" : "Unpinned"}
      </p>
    </div>
  );
}
```
