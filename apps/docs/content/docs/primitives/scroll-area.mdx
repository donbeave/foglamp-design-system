## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `ScrollArea` | `ScrollAreaPrimitive.Root.Props` | Scroll container root. |
| `ScrollBar` | `ScrollAreaPrimitive.Scrollbar.Props` | Scrollbar for the scroll area. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `ScrollBar.orientation` | `vertical \| horizontal` | Defaults to `vertical`. |

## Usage

```tsx
import { ScrollArea, ScrollBar } from "@tailrocks/framerock/ui/scroll-area";

export function ScrollAreaExample({
  itemCount = 8,
  height = 180,
}: {
  itemCount?: number;
  height?: number;
} = {}) {
  return (
    <ScrollArea
      style={{ height }}
      className="w-full max-w-sm rounded-md shadow-(--custom-shadow)"
    >
      <div className="space-y-3 p-4">
        {Array.from({ length: Math.max(1, itemCount) }, (_, i) => (
            <div
              key={i}
              className="flex justify-between border-b pb-2 text-sm"
            >
              <span>
                trace_
                {String(i + 1).padStart(3, "0")}
                · generation
              </span>
              <span className="text-muted-foreground">
                {120 + i * 17}
                ms
              </span>
            </div>
        ))}
      </div>
      <ScrollBar orientation="vertical" />
    </ScrollArea>
  );
}
```
