## API and props


| Export | Declared props or type | Role |
| --- | --- | --- |
| `Pagination` | `React.ComponentProps<"nav">` | Navigation landmark for page controls. |
| `PaginationContent` | `React.ComponentProps<"ul">` | Contains the associated popup or panel content. |
| `PaginationEllipsis` | `React.ComponentProps<"span">` | Truncation marker for omitted pages or content. |
| `PaginationItem` | `React.ComponentProps<"li">` | One selectable or repeated entry. |
| `PaginationLink` | `PaginationLinkProps` | Page link with active state and Button size. |
| `PaginationNext` | `React.ComponentProps<typeof PaginationLink> & { text?: string }` | Next-item or next-page control. |
| `PaginationPrevious` | `React.ComponentProps<typeof PaginationLink> & { text?: string }` | Previous-item or previous-page control. |

## Additional props

| Prop | Type | Description |
| --- | --- | --- |
| `PaginationLink.isActive` | `boolean` | Marks the current page. |
| `PaginationLink.size` | `Button size` | Defaults to `icon-sm`. |
| `PaginationPrevious.text` | `string` | Defaults to `Previous`. |
| `PaginationNext.text` | `string` | Defaults to `Next`. |

## Usage

```tsx
import {
  Pagination,
  PaginationContent,
  PaginationItem,
  PaginationLink,
  PaginationPrevious,
  PaginationNext,
  PaginationEllipsis,
} from "@tailrocks/framerock/ui/pagination";

export function PaginationExample({
  currentPage = 2,
  showEdges = true,
}: {
  currentPage?: number;
  showEdges?: boolean;
} = {}) {
  return (
    <Pagination>
      <PaginationContent>
        {showEdges ? (
            <PaginationItem>
              <PaginationPrevious href="#previous" />
            </PaginationItem>
        ) : null}
        {[1, 2, 3].map((page) => (
            <PaginationItem key={page}>
              <PaginationLink
                href={`#${page}`}
                isActive={currentPage === page}
              >
                {page}
              </PaginationLink>
            </PaginationItem>
        ))}
        <PaginationItem>
          <PaginationEllipsis />
        </PaginationItem>
        {showEdges ? (
            <PaginationItem>
              <PaginationNext href="#next" />
            </PaginationItem>
        ) : null}
      </PaginationContent>
    </Pagination>
  );
}
```
