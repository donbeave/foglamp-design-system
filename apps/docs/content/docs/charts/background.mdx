## Public API

| Export | Prop / Type | Type | Description |
| :--- | :--- | :--- | :--- |
| `ChartBackground` | `variant` | `BackgroundVariant` | SVG pattern rendered behind the chart plot area. |
| `BackgroundVariant` | `type` | `"dots" \| "grid" \| "cross-hatch" \| "diagonal-lines" \| "plus" \| "falling-triangles" \| "4-pointed-star" \| "tiny-checkers" \| "overlapping-circles" \| "wiggle-lines" \| "bubbles"` | Supported background pattern identifiers. |

## Requirements

`BackgroundVariant` accepts `dots`, `grid`, `cross-hatch`, `diagonal-lines`, `plus`, `falling-triangles`, `four-pointed-star`, `tiny-checkers`, `overlapping-circles`, `wiggle-lines`, and `bubbles`.

## Usage

```tsx
import {
  EvilAreaChart,
  Area,
  Grid,
  Tooltip,
  XAxis,
  YAxis,
} from "@tailrocks/framerock/charts/evilcharts/charts/area-chart";
import {
  BackgroundVariant,
} from "@tailrocks/framerock/charts/evilcharts/ui/background";
import {
  ChartBackground,
} from "@tailrocks/framerock/charts/evilcharts/ui/background";
import type { ChartConfig } from "@tailrocks/framerock/charts/evilcharts/ui/chart";

const observabilityConfig = {
  requests: { label: "Requests", colors: { light: ["#6d5efc"], dark: ["#a59bff"] } },
  errors: { label: "Errors", colors: { light: ["#e05d71"], dark: ["#ff8292"] } },
  latency: { label: "Latency", colors: { light: ["#19a98c"], dark: ["#54d9b5"] } },
} satisfies ChartConfig;

const observabilityData = [
  { day: "Mon", requests: 116, errors: 9, latency: 178 },
  { day: "Tue", requests: 132, errors: 12, latency: 194 },
  { day: "Wed", requests: 124, errors: 7, latency: 169 },
  { day: "Thu", requests: 158, errors: 15, latency: 218 },
  { day: "Fri", requests: 147, errors: 11, latency: 203 },
];

export function BackgroundExample({
  variant = 'dots',
}: {
  variant?: BackgroundVariant;
} = {}) {
  return (
    <EvilAreaChart
      config={observabilityConfig}
      data={observabilityData}
      className="h-[300px]"
      animationType="none"
    >
      <ChartBackground variant={variant} />
      <Grid />
      <XAxis dataKey="day" />
      <YAxis width={32} />
      <Tooltip />
      <Area dataKey="requests" variant="gradient" />
    </EvilAreaChart>
  );
}
```
