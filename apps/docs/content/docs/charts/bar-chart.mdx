## Requirements

Bar series use `dataKey` values present in both the rows and `config`. `layout` controls bar orientation. `stackType` accepts `default`, `stacked`, or `percent`. Give the chart container a measurable height.

## Usage

```tsx
import {
  EvilBarChart,
  Bar,
  Grid,
  Legend,
  Threshold,
  Tooltip,
  XAxis,
  YAxis,
} from "@tailrocks/framerock/charts/evilcharts/charts/bar-chart";
import {
  ChartBackground,
} from "@tailrocks/framerock/charts/evilcharts/ui/background";
import type { ChartConfig } from "@tailrocks/framerock/charts/evilcharts/ui/chart";

const observabilityConfig = {
  requests: { label: "Requests", colors: { light: ["#6d5efc"], dark: ["#a59bff"] } },
  errors: { label: "Errors", colors: { light: ["#e05d71"], dark: ["#ff8292"] } },
  latency: { label: "Latency", colors: { light: ["#19a98c"], dark: ["#54d9b5"] } },
} satisfies ChartConfig;

const observabilityData = [
  { day: "Mon", requests: 116, errors: 9, latency: 178 },
  { day: "Tue", requests: 132, errors: 12, latency: 194 },
  { day: "Wed", requests: 124, errors: 7, latency: 169 },
  { day: "Thu", requests: 158, errors: 15, latency: 218 },
  { day: "Fri", requests: 147, errors: 11, latency: 203 },
];

export function BarExample({
  layout = 'vertical',
  stacked = false,
}: {
  layout?: 'vertical' | 'horizontal';
  stacked?: boolean;
} = {}) {
  return (
    <EvilBarChart
      config={observabilityConfig}
      data={observabilityData}
      className="h-[300px]"
      layout={layout}
      stackType={stacked ? 'stacked' : 'default'}
    >
      <ChartBackground variant="grid" />
      <Grid />
      <XAxis
        dataKey={layout === 'vertical' ? 'day' : undefined}
        type={layout === 'horizontal' ? 'number' : 'category'}
      />
      <YAxis
        dataKey={layout === 'horizontal' ? 'day' : undefined}
        type={layout === 'horizontal' ? 'category' : 'number'}
        width={layout === 'horizontal' ? 52 : 36}
      />
      <Tooltip variant="frosted-glass" />
      <Legend align="left" isClickable />
      <Threshold value={200} label="request budget" />
      <Bar dataKey="requests" />
      <Bar dataKey="errors" />
    </EvilBarChart>
  );
}
```
