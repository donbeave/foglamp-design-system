## Requirements

Area series use `dataKey` values present in both the rows and `config`. Area variants include `gradient`, `hatched`, and `solid`; `stackType` controls stacking. Give the chart container a measurable height.

## Usage

```tsx
import {
  EvilAreaChart,
  Area,
  Dot,
  Grid,
  Legend,
  Tooltip,
  XAxis,
  YAxis,
} from "@tailrocks/framerock/charts/evilcharts/charts/area-chart";
import {
  ChartBackground,
} from "@tailrocks/framerock/charts/evilcharts/ui/background";
import type { ChartConfig } from "@tailrocks/framerock/charts/evilcharts/ui/chart";

const observabilityConfig = {
  requests: { label: "Requests", colors: { light: ["#6d5efc"], dark: ["#a59bff"] } },
  errors: { label: "Errors", colors: { light: ["#e05d71"], dark: ["#ff8292"] } },
  latency: { label: "Latency", colors: { light: ["#19a98c"], dark: ["#54d9b5"] } },
} satisfies ChartConfig;

const observabilityData = [
  { day: "Mon", requests: 116, errors: 9, latency: 178 },
  { day: "Tue", requests: 132, errors: 12, latency: 194 },
  { day: "Wed", requests: 124, errors: 7, latency: 169 },
  { day: "Thu", requests: 158, errors: 15, latency: 218 },
  { day: "Fri", requests: 147, errors: 11, latency: 203 },
];

export function AreaExample({
  variant = 'gradient',
  isLoading = false,
}: {
  variant?: 'gradient' | 'hatched' | 'solid';
  isLoading?: boolean;
} = {}) {
  return (
    <EvilAreaChart
      config={observabilityConfig}
      data={observabilityData}
      className="h-[300px]"
      isLoading={isLoading}
      curveType="monotone"
      animationType="left-to-right"
    >
      <ChartBackground variant="dots" />
      <Grid />
      <XAxis dataKey="day" />
      <YAxis width={32} />
      <Tooltip variant="frosted-glass" />
      <Legend align="left" isClickable />
      <Area dataKey="requests" variant={variant} isClickable>
        <Dot variant="border" />
      </Area>
      <Area dataKey="errors" variant="gradient" isClickable />
    </EvilAreaChart>
  );
}
```
