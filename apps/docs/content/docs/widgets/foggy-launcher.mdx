## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `FoggyLauncher` | `onOpen` | `() => void` | Runs when the launcher opens the assistant. |
|  | `showButton` | `boolean \| undefined` | Shows the launcher button. |

## Usage

```tsx
import { useState } from "react";
import {
  FoggyLauncher,
} from "@tailrocks/framerock/widgets/foggy/foggy-widget";

export function FoggyLauncherExample({
  showButton = true,
}: {
  showButton?: boolean;
} = {}) {
  const [open, setOpen] = useState(false);
  return (
    <div className="relative h-40 w-full rounded-lg border bg-muted/20">
      <div className="absolute bottom-3 right-3">
        <FoggyLauncher
          onOpen={() => setOpen(!open)}
          showButton={showButton}
        />
      </div>
      <p className="absolute bottom-4 left-4 text-xs text-muted-foreground">
        Assistant
        {open ? 'opened' : 'closed'}
      </p>
    </div>
  );
}
```
