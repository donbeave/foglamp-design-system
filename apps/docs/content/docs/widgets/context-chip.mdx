## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `ContextChip` | `href` | `string \| undefined` | Optional link destination. |
|  | `icon` | `ComponentType<{ className?: string; }>` | Icon component rendered beside the label. |
|  | `iconClassName` | `string \| undefined` | Adds a class to the leading icon. |
|  | `label` | `string` | Visible label text. |

## Usage

```tsx
import { ContextChip } from "@tailrocks/framerock/widgets/context-chip";

function ChipMark({ className }: { className?: string }) {
  return (
    <span className={className} aria-hidden="true">●</span>
  );
}

export function ContextChipExample({
  label = 'Support agent',
}: {
  label?: string;
} = {}) {
  return (
    <div className="flex flex-wrap gap-2">
      <ContextChip icon={ChipMark} label={label} />
      <ContextChip icon={ChipMark} label="Last 7 days" href="/traces" />
    </div>
  );
}
```
