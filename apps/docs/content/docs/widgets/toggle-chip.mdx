## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `ToggleChip` | `active` | `boolean` | Whether the filter chip is selected. |
|  | `onClick` | `() => void` | Runs when the user activates the control. |
|  | `children` | `ReactNode` | Content rendered inside the component. |

## Usage

```tsx
import { useState } from "react";
import { ToggleChip } from "@tailrocks/framerock/widgets/data-table";

export function ToggleChipExample({
  initial = false,
}: {
  initial?: boolean;
} = {}) {
  const [active, setActive] = useState(initial);
  return (
    <ToggleChip active={active} onClick={() => setActive(!active)}>
      Errors only
    </ToggleChip>
  );
}
```
