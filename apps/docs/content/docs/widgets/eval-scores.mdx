## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `ScoreRow` | `score` | `TraceScore` | Evaluation score displayed in this row. |
|  | `meta` | `EvalMeta \| undefined` | Evaluation name and preset metadata for the score. |
|  | `presetName` | `Map<string, string>` | Preset names keyed by preset ID. |
| `SpanScoreDots` | `scores` | `TraceScore[]` | Evaluation scores associated with trace spans. |
|  | `evalMeta` | `Map<string, EvalMeta>` | Evaluation names keyed by evaluation ID. |
|  | `presetName` | `Map<string, string>` | Preset names keyed by preset ID. |
|  | `max` | `number \| undefined` | Sets the upper bound for the displayed score marks. |
|  | `className` | `string \| undefined` | Adds a class to the rendered root element. |

## Usage

```tsx
import {
  ScoreRow,
  SpanScoreDots,
  type EvalMeta,
  type TraceScore,
} from "@tailrocks/framerock/widgets/eval-scores";

const scores: TraceScore[] = [
  {
    scoreId: "score-1",
    evalId: "eval-accuracy",
    targetType: "trace",
    targetId: "trace-01",
    traceId: "trace-01",
    scorer: "llm",
    label: "Answer relevance",
    score: 0.94,
    passed: true,
    reason: "Grounded in retrieved billing documentation.",
    modelId: "gpt-4o",
    cost: 0.0004,
    scoredAt: "2025-02-14T10:01:00Z",
  },
];

const meta = new Map<string, EvalMeta>([
  [
    "eval-accuracy",
    {
      name: "Answer relevance",
      presetId: "relevance",
    },
  ],
]);

const presetName = new Map<string, string>([
  ["relevance", "Answer Relevance"],
]);

export function EvalScoresExample() {
  return (
    <div className="space-y-3">
      <div className="flex items-center gap-2 text-xs text-muted-foreground">
        <span>Span indicators</span>
        <SpanScoreDots
          scores={scores}
          evalMeta={meta}
          presetName={presetName}
        />
      </div>
      <div className="space-y-2">
        {scores.map((score) => (
          <ScoreRow
            key={score.scoreId}
            score={score}
            meta={meta.get(score.evalId)}
            presetName={presetName}
          />
        ))}
      </div>
    </div>
  );
}
```
