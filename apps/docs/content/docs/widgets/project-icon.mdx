## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `ProjectIcon` | `url` | `string \| null \| undefined` | Optional image URL for the project mark. |
|  | `name` | `string \| null \| undefined` | Name used to select the rendered label or icon. |
|  | | `size` | `"xs" \| "sm" \| "md" \| undefined` | Project badge size preset (`"xs"` \| `"sm"` \| `"md"`). |

## Usage

```tsx
import { ProjectIcon } from "@tailrocks/framerock/widgets/leaf/project-icon";

export function ProjectIconExample({
  name = 'Acme Production',
  size = 'md',
}: {
  name?: string;
  size?: 'xs' | 'sm' | 'md';
} = {}) {
  return (
    <div className="flex items-center gap-3">
      <ProjectIcon name={name} url={null} size={size} />
      <span>
        {name}
      </span>
    </div>
  );
}
```
