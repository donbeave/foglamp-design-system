## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `FilterSelect` | `value` | `"" \| T` | Controlled value or source data passed to the component. |
|  | `onChange` | `(value: T \| "") => void` | Receives the next value when the user changes the control. |
|  | `allLabel` | `string` | Label for the unfiltered option. |
|  | `options` | `{ value: T; label: string; hint?: string; icon?: ComponentType<{ className?: string; }>; }[]` | Selectable values and their display labels. |
|  | `icon` | `ComponentType<{ className?: string; }> \| undefined` | Icon component rendered beside the label. |
|  | `className` | `string \| undefined` | Adds a class to the rendered root element. |
|  | `allowFreeText` | `boolean \| undefined` | Allows values outside the listed options. |

## Usage

```tsx
import { useState } from "react";
import { FilterSelect } from "@tailrocks/framerock/widgets/data-table";

const statusOptions: { value: string; label: string }[] = [
  { value: 'completed', label: 'Completed' },
  { value: 'running', label: 'Running' },
  { value: 'failed', label: 'Failed' },
  { value: 'queued', label: 'Queued' },
  { value: 'cancelled', label: 'Cancelled' },
  { value: 'timed_out', label: 'Timed out' },
  { value: 'retrying', label: 'Retrying' },
  { value: 'waiting', label: 'Waiting' },
  { value: 'paused', label: 'Paused' },
];

export function FilterSelectExample({
  allLabel = 'Any status',
}: {
  allLabel?: string;
} = {}) {
  const [value, setValue] = useState<string>('completed');
  return (
    <FilterSelect
      value={value}
      onChange={setValue}
      allLabel={allLabel}
      options={statusOptions}
    />
  );
}
```
