## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `RelativeTime` | `value` | `string \| Date \| null \| undefined` | Controlled value or source data passed to the component. |

## Usage

```tsx
import {
  RelativeTime,
} from "@tailrocks/framerock/widgets/leaf/relative-time";

export function RelativeTimeExample({
  value = '2025-02-14T10:00:00.000Z',
}: {
  value?: string | null;
} = {}) {
  return (
    <span className="text-sm">
      <RelativeTime value={value} />
    </span>
  );
}
```
