## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `DevBarView` | `fps` | `number \| undefined` | Frames per second shown by the developer bar. |
|  | `queries` | `{ fetching: QueryDiagnostic[]; stale: QueryDiagnostic[]; staleInactive: number; } \| undefined` | Fetching, stale, and inactive query diagnostics. |
|  | `netDelay` | `number \| undefined` | Simulated network delay in milliseconds. |
|  | `netFail` | `boolean \| undefined` | Shows a simulated network failure. |
|  | `forceLoading` | `boolean \| undefined` | Forces loading indicators on. |
|  | `onNetworkDelayChange` | `((delay: DevNetworkDelay) => void) \| undefined` | Receives the selected network delay. |
|  | `onNetworkFailChange` | `((fail: boolean) => void) \| undefined` | Receives the network failure setting. |
|  | `onForceLoadingChange` | `((loading: boolean) => void) \| undefined` | Receives the forced-loading setting. |
|  | `onInvalidateQueries` | `(() => void) \| undefined` | Requests query invalidation from the host application. |
|  | `onClearQueries` | `(() => void) \| undefined` | Clears query diagnostics. |
|  | `onSendTestEmail` | `((variant: string) => void) \| undefined` | Receives the selected test email variant. |
|  | `sendingTestEmail` | `boolean \| undefined` | Marks a test email as sending. |

## Requirements

This view receives diagnostic snapshots and host actions as props. Supply `queries` and `fps`, then pass callbacks such as `onInvalidateQueries` for host actions. It does not fetch diagnostics itself.

## Usage

```tsx
import { useState } from "react";
import {
  DevBarView,
  type DevNetworkDelay,
} from "@tailrocks/framerock/widgets/dev-toolbar";

export function DevBarExample({ initialForceLoading = false, initialNetFail = false } = {}) {
  const [netDelay, setNetDelay] = useState<DevNetworkDelay>(0);
  const [netFail, setNetFail] = useState(initialNetFail);
  const [forceLoading, setForceLoading] = useState(initialForceLoading);
  const [invalidatedCount, setInvalidatedCount] = useState(0);
  const [clearedCount, setClearedCount] = useState(0);
  const [queries, setQueries] = useState({
    fetching: [{ hash: 'q_83a1', label: 'traces.list' }],
    stale: [{ hash: 'q_829a', label: 'projects.active' }],
    staleInactive: 2,
  });
  return (
    <div className="relative min-h-48 w-full max-w-xl">
      {invalidatedCount > 0 || clearedCount > 0 ? (
          <div className="mb-2 text-xs text-muted-foreground" role="status">
            {invalidatedCount > 0 ? `Invalidated queries (${invalidatedCount})` : ''}
            {invalidatedCount > 0 && clearedCount > 0 ? ' · ' : ''}
            {clearedCount > 0 ? `Cleared cache (${clearedCount})` : ''}
          </div>
      ) : null}
      <DevBarView
        fps={60}
        queries={queries}
        netDelay={netDelay}
        netFail={netFail}
        forceLoading={forceLoading}
        onNetworkDelayChange={setNetDelay}
        onNetworkFailChange={setNetFail}
        onForceLoadingChange={setForceLoading}
        onInvalidateQueries={() => { setInvalidatedCount((c) => c + 1); setQueries({ fetching: [], stale: [], staleInactive: 0, }); window.dispatchEvent(new Event('framerock-story:invalidate')); }}
        onClearQueries={() => { setClearedCount((c) => c + 1); setQueries({ fetching: [], stale: [], staleInactive: 0, }); window.dispatchEvent(new Event('framerock-story:clear')); }}
      />
    </div>
  );
}
```
