## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `ModelChip` | `models` | `ChipModel[]` | Model IDs displayed in the chip. |

## Usage

```tsx
import { ModelChip } from "@tailrocks/framerock/widgets/model-chip";

export function ModelChipExample({
  multi = false,
}: {
  multi?: boolean;
} = {}) {
  const models = [{ modelId: 'gpt-4o', provider: 'openai' }, ...(multi ? [{ modelId: 'claude-3-7-sonnet', provider: 'anthropic' }, { modelId: 'gemini-2.5-pro', provider: 'google' }] : [])];
  return <ModelChip models={models} />;
}
```
