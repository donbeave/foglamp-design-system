## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `PromptVersionChip` | `version` | `{ id: string; number: number; } \| null \| undefined` | Prompt version associated with the link. |
|  | `agentName` | `string \| null \| undefined` | Agent name used to build the prompt versions link. |
|  | `className` | `string \| undefined` | Adds a class to the rendered root element. |

## Usage

```tsx
import {
  PromptVersionChip,
} from "@tailrocks/framerock/widgets/prompt-version-chip";

export function PromptVersionChipExample({
  linked = true,
}: {
  linked?: boolean;
} = {}) {
  return (
    <PromptVersionChip
      version={{ id: 'v12', number: 12 }}
      agentName={linked ? 'Support Assistant' : null}
    />
  );
}
```
