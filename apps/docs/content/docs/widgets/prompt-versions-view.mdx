## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `PromptVersionsView` | `versions` | `PromptVersion[]` | Prompt versions rendered by the comparison view. |
|  | `loading` | `boolean` | Controls the loading state. |
|  | `skeleton` | `boolean` | Shows skeleton placeholders when true. |
|  | `className` | `string \| undefined` | Adds a class to the rendered root element. |
|  | `renderProse` | `((version: PromptVersion) => ReactNode) \| undefined` | Renders a version template as prose. |

## Usage

```tsx
import {
  PromptVersionsView,
} from "@tailrocks/framerock/widgets/prompt-versions-card";

const promptVersionsFixture = [
  {
    id: "v1",
    number: 1,
    template: "You are a helpful support assistant.\n{{customer_context}}",
    slotCount: 1,
    hashCount: 3,
    runCount: 128,
    firstSeen: "2025-02-01T08:00:00Z",
    lastSeen: "2025-02-08T12:00:00Z",
    current: false,
  },
  {
    id: "v2",
    number: 2,
    template: "You are a concise support assistant.\n{{customer_context}}",
    slotCount: 1,
    hashCount: 5,
    runCount: 84,
    firstSeen: "2025-02-08T12:00:00Z",
    lastSeen: "2025-02-14T10:00:00Z",
    current: true,
  },
];

export function PromptVersionsExample({
  loading = false,
}: {
  loading?: boolean;
} = {}) {
  return (
    <div className="w-full">
      <PromptVersionsView
        versions={promptVersionsFixture}
        loading={loading}
        skeleton={loading}
      />
    </div>
  );
}
```
