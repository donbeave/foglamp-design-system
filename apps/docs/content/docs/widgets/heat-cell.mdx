## Public API


| Export | Prop | Type | Meaning |
| :--- | :--- | :--- | :--- |
| `HeatCell` | `value` | `number \| null \| undefined` | Controlled value or source data passed to the component. |
|  | `thresholds` | `number[]` | Threshold values used to shade the cell. |
|  | `metric` | `"cost" \| "duration" \| "spend" \| undefined` | Metric displayed in the cell tooltip. |
|  | `bold` | `boolean \| undefined` | Uses bold text for the value. |
|  | `mutedWhenZero` | `boolean \| undefined` | Mutes the cell when its value is zero. |
|  | `children` | `ReactNode` | Content rendered inside the component. |

## Usage

```tsx
import {
  Table,
  TableBody,
  TableCell,
  TableRow,
} from "@tailrocks/framerock/ui/table";
import { HeatCell } from "@tailrocks/framerock/widgets/heat-cell";

export function HeatCellExample() {
  return (
    <Table className="w-72">
      <TableBody>
        <TableRow>
          <TableCell>Fastest</TableCell>
          <HeatCell
            value={42}
            thresholds={[80, 160, 280, 500]}
            metric="duration"
          >
            42 ms
          </HeatCell>
        </TableRow>
        <TableRow>
          <TableCell>Slowest</TableCell>
          <HeatCell
            value={720}
            thresholds={[80, 160, 280, 500]}
            metric="duration"
            bold
          >
            720 ms
          </HeatCell>
        </TableRow>
      </TableBody>
    </Table>
  );
}
```
